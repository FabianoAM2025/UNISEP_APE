        salario_final = salario * 0.05f;

    } else {
        salario_final = salario * 0.10f;

5)Verificação de Maioridade para Votação
Leia a idade e informe:

Não pode votar (< 16)
Voto opcional (16–17 ou > 70)
Voto obrigatório (18–70)
----------------------------------------------------------
#include <stdio.h>

int main(){


int idade = 0;

    printf("Digite sua idade:\n");
    scanf("%i", &idade);

    if (idade <= 12) {
        printf("Voce e uma Crianca\n");
    } else if (idade >= 13 && idade <= 17) {
        printf("Voce e um Adolescente\n");
    } else if (idade >= 18 && idade <= 59) {
        printf("Voce e um Adulto\n");
    } else {
        printf("Voce e um Idoso\n");
    }

    return 0;
}
